# Underwriter v1 — dependency manifest
#
# Each dependency has a single job. Pinned to a minor floor rather than an
# exact version: ESPN changes its private API without notice, and the espn-api
# wrapper is the layer that absorbs those changes, so it needs room to move
# within a major version.

# Desktop shell. Hosts the war-room UI in a native macOS WebKit window and
# bridges it to the Python backend through a js_api object. Chosen over Electron
# because the engine is already Python and this avoids shipping a second runtime.
pywebview>=5.0,<6.0

# Drives a real Chrome binary for ESPN session-credential capture.
# Only the auth path imports this; the rest of the engine is pure HTTP.
playwright>=1.40,<2.0

# Typed wrapper over ESPN's private v3 fantasy API. Supplies league settings,
# rosters, and the free-agent pool.
espn-api>=0.35,<1.0

# Frames the draftable pool for the value-over-replacement pass. Pulls numpy in
# transitively, which is most of the bundle weight.
pandas>=2.0,<3.0

# Loads SWID / espn_s2 / LEAGUE_ID from the local .env. Credentials never live
# in source, and .env is gitignored.
python-dotenv>=1.0,<2.0

# Direct transport for the live-draft hot path, where we need control over the
# x-fantasy-filter header and cannot afford a full league re-fetch per poll.
requests>=2.31,<3.0

# Build-time only. Produces the distributable macOS .app bundle.
# Install with: pip install pyinstaller
# pyinstaller>=6.0,<7.0
